একটি নতুন ট্যাবে গুগল অ্যাপস স্ক্রিপ্ট এডিটর ওপেন হবে।

৩
নিচের স্ক্রিপ্ট কোডটি কপি ও পেস্ট করুন

Apps Script এডিটরের Code.gs ফাইলে থাকা সব লেখা মুছে ফেলুন (Ctrl+A চেপে Delete দিন)।

এরপর নিচের কোডটি Copy Script Code বাটনে ক্লিক করে কপি করে সম্পূর্ণ পেস্ট করে দিন:

Bookly_Backup_AppScript.gs 
/**
 * BOOKLY PRO - GOOGLE APPS SCRIPT BACKUP & 2-WAY LIVE SYNC ENGINE
 * Version: 4.1.0
 */

var TAB_BOOKMARKS = "Bookmarks";
var TAB_FOLDERS = "Folders";
var TAB_QUICK_ACCESS = "Quick Access";
var TAB_RAW_BACKUP = "_raw_snapshot";

function doGet(e) { return handleRequest(e); }
function doPost(e) { return handleRequest(e); }

function handleRequest(e) {
  try {
    var params = (e && e.parameter) ? e.parameter : {};
    var postData = {};
    if (e && e.postData && e.postData.contents) {
      try { postData = JSON.parse(e.postData.contents); } catch (err) {}
    }
    var data = Object.assign({}, params, postData);
    var action = (data.action || '').trim().toLowerCase();

    if (!action || action === 'dashboard') return renderDashboard();

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (action === 'backup' || action === 'sync_push') return handleBackup(ss, data);
    else if (action === 'restore' || action === 'get_backup' || action === 'sync_pull') return handleRestore(ss, data);
    else if (action === 'stats' || action === 'ping') return handleStats(ss);
    else return jsonResponse({ success: false, code: 'UNKNOWN_ACTION', message: 'Action not supported.' });
  } catch (err) {
    return jsonResponse({ success: false, code: 'SERVER_ERROR', message: err.message || String(err) });
  }
}

function handleBackup(ss, data) {
  var now = new Date();
  var timestampIso = now.toISOString();
  var links = (data.links && typeof data.links === 'object') ? data.links : {};
  var folderOrder = Array.isArray(data.folderOrder) ? data.folderOrder : [];
  var pinnedFolders = Array.isArray(data.pinnedFolders) ? data.pinnedFolders : [];
  var favoriteFolders = (data.favoriteFolders && typeof data.favoriteFolders === 'object') ? data.favoriteFolders : {};
  var folderIcons = (data.folderIcons && typeof data.folderIcons === 'object') ? data.folderIcons : {};
  var quickAccess = Array.isArray(data.quickAccess) ? data.quickAccess : [];

  var totalBookmarks = countTotalBookmarks(links);
  if (totalBookmarks === 0) {
    var existingRaw = readRawBackup(ss);
    if (existingRaw && countTotalBookmarks(existingRaw.links) > 0) {
      return jsonResponse({
        success: false, code: 'CLOUD_PROTECTED',
        message: 'Empty backup rejected: Google Sheet already has bookmarks. Protected.'
      });
    }
  }

  var totalBookmarksUpdated = updateBookmarksSheet(ss, links);
  var totalFoldersUpdated = updateFoldersSheet(ss, links, folderOrder, pinnedFolders, favoriteFolders, folderIcons);
  updateQuickAccessSheet(ss, quickAccess);

  var rawPayload = {
    version: data.version || "4.1.0",
    exportDate: timestampIso,
    links: links,
    folderOrder: folderOrder,
    pinnedFolders: pinnedFolders,
    favoriteFolders: favoriteFolders,
    folderIcons: folderIcons,
    quickAccess: quickAccess,
    customLinkOrder: data.customLinkOrder || {},
    theme: data.theme || "dark",
    dailyFocusTasks: data.dailyFocusTasks || [],
    dailyNotes: data.dailyNotes || "",
    todoWindowPosition: data.todoWindowPosition || null
  };
  saveRawBackup(ss, rawPayload);

  return jsonResponse({
    success: true, code: 'BACKUP_SUCCESS',
    message: 'Bookly Pro backup successfully saved to Google Sheets!',
    timestamp: timestampIso,
    counts: { bookmarks: totalBookmarksUpdated, folders: totalFoldersUpdated, quickAccess: quickAccess.length }
  });
}

function handleRestore(ss, data) {
  var rawData = readRawBackup(ss);
  if (rawData && rawData.links && Object.keys(rawData.links).length > 0) {
    return jsonResponse({ success: true, code: 'RESTORE_SUCCESS', source: 'raw_snapshot', backup: rawData, totalBookmarks: countTotalBookmarks(rawData.links) });
  }
  var reconstructed = reconstructFromSheets(ss);
  return jsonResponse({ success: true, code: 'RESTORE_SUCCESS', source: 'sheets_table', backup: reconstructed, totalBookmarks: countTotalBookmarks(reconstructed.links) });
}

function handleStats(ss) {
  var rawData = readRawBackup(ss);
  var totalBookmarks = rawData && rawData.links ? countTotalBookmarks(rawData.links) : 0;
  var totalFolders = rawData && rawData.links ? Object.keys(rawData.links).length : 0;
  return jsonResponse({
    success: true, code: 'PONG', sheetName: ss.getName(),
    totalBookmarks: totalBookmarks, totalFolders: totalFolders, lastBackupAt: rawData ? rawData.exportDate : null
  });
}

function updateBookmarksSheet(ss, links) {
  var sheet = getOrCreateSheet(ss, TAB_BOOKMARKS);
  sheet.clearContents();
  var headers = ['Folder', 'Title', 'URL', 'Tags', 'Notes', 'Favorite', 'Pinned', 'Order', 'ID', 'Date Added'];
  var rows = [headers];
  var total = 0;
  for (var folder in links) {
    if (!links.hasOwnProperty(folder)) continue;
    var list = links[folder];
    if (!Array.isArray(list)) continue;
    for (var i = 0; i < list.length; i++) {
      var b = list[i]; if (!b) continue; total++;
      var tags = Array.isArray(b.tags) ? b.tags.join(', ') : (b.tags || '');
      rows.push([folder, b.title || '', b.link || '', tags, b.note || b.notes || '', b.favorite ? 'Yes' : 'No', b.pinned ? 'Yes' : 'No', b.order !== undefined ? b.order : (i * 10), b.id || '', b.dateAdded || '']);
    }
  }
  if (rows.length > 0) {
    sheet.getRange(1, 1, rows.length, headers.length).setValues(rows);
    formatHeaderRow(sheet, headers.length, '#4f46e5');
    sheet.setFrozenRows(1); sheet.setFrozenColumns(1);
  }
  return total;
}

function updateFoldersSheet(ss, links, folderOrder, pinnedFolders, favoriteFolders, folderIcons) {
  var sheet = getOrCreateSheet(ss, TAB_FOLDERS);
  sheet.clearContents();
  var headers = ['Folder Name', 'Order Sequence', 'Total Bookmarks', 'Pinned', 'Favorite', 'Icon / Emoji'];
  var rows = [headers];
  var allFolders = []; var seen = {};
  if (Array.isArray(folderOrder)) {
    for (var i = 0; i < folderOrder.length; i++) {
      var f = folderOrder[i]; if (f && !seen[f]) { seen[f] = true; allFolders.push(f); }
    }
  }
  for (var fName in links) {
    if (links.hasOwnProperty(fName) && !seen[fName]) { seen[fName] = true; allFolders.push(fName); }
  }
  for (var idx = 0; idx < allFolders.length; idx++) {
    var name = allFolders[idx];
    rows.push([name, idx, Array.isArray(links[name]) ? links[name].length : 0, pinnedFolders.indexOf(name) !== -1 ? 'Yes' : 'No', !!favoriteFolders[name] ? 'Yes' : 'No', folderIcons[name] || '']);
  }
  if (rows.length > 0) {
    sheet.getRange(1, 1, rows.length, headers.length).setValues(rows);
    formatHeaderRow(sheet, headers.length, '#059669');
    sheet.setFrozenRows(1); sheet.setFrozenColumns(1);
  }
  return allFolders.length;
}

function updateQuickAccessSheet(ss, quickAccess) {
  var sheet = getOrCreateSheet(ss, TAB_QUICK_ACCESS);
  sheet.clearContents();
  var headers = ['Title', 'URL', 'Icon / Favicon', 'Order'];
  var rows = [headers];
  if (Array.isArray(quickAccess)) {
    for (var i = 0; i < quickAccess.length; i++) {
      var item = quickAccess[i]; if (!item) continue;
      rows.push([item.title || '', item.url || item.link || '', item.icon || '', i]);
    }
  }
  if (rows.length > 0) {
    sheet.getRange(1, 1, rows.length, headers.length).setValues(rows);
    formatHeaderRow(sheet, headers.length, '#d97706');
    sheet.setFrozenRows(1);
  }
}

function saveRawBackup(ss, rawPayload) {
  var sheet = getOrCreateSheet(ss, TAB_RAW_BACKUP);
  sheet.clearContents();
  var json = JSON.stringify(rawPayload);
  var chunkSize = 45000; var chunks = [];
  for (var i = 0; i < json.length; i += chunkSize) chunks.push([json.substring(i, i + chunkSize)]);
  if (chunks.length > 0) sheet.getRange(1, 1, chunks.length, 1).setValues(chunks);
  try { sheet.hideSheet(); } catch (_) {}
}

function readRawBackup(ss) {
  var sheet = ss.getSheetByName(TAB_RAW_BACKUP);
  if (!sheet || sheet.getLastRow() < 1) return null;
  var values = sheet.getRange(1, 1, sheet.getLastRow(), 1).getValues();
  var json = '';
  for (var i = 0; i < values.length; i++) json += String(values[i][0] || '');
  try { return JSON.parse(json); } catch (e) { return null; }
}

function reconstructFromSheets(ss) {
  var links = {}; var folderOrder = []; var quickAccess = [];
  var bookSheet = ss.getSheetByName(TAB_BOOKMARKS);
  if (bookSheet && bookSheet.getLastRow() > 1) {
    var data = bookSheet.getRange(2, 1, bookSheet.getLastRow() - 1, 10).getValues();
    for (var i = 0; i < data.length; i++) {
      var r = data[i]; var folder = String(r[0] || 'General').trim() || 'General';
      links[folder] = links[folder] || [];
      links[folder].push({
        id: String(r[8] || ('b_' + Date.now() + '_' + i)), title: String(r[1] || ''), link: String(r[2] || ''),
        tags: String(r[3] || '').split(',').map(function(t) { return t.trim(); }).filter(Boolean),
        note: String(r[4] || ''), favorite: String(r[5]).toLowerCase() === 'yes', pinned: String(r[6]).toLowerCase() === 'yes',
        order: Number(r[7]) || (i * 10), dateAdded: String(r[9] || new Date().toISOString())
      });
    }
  }
  var foldSheet = ss.getSheetByName(TAB_FOLDERS);
  if (foldSheet && foldSheet.getLastRow() > 1) {
    var foldData = foldSheet.getRange(2, 1, foldSheet.getLastRow() - 1, 2).getValues();
    for (var j = 0; j < foldData.length; j++) {
      var f = String(foldData[j][0] || '').trim(); if (f) folderOrder.push(f);
    }
  }
  return { version: "4.1.0", exportDate: new Date().toISOString(), links: links, folderOrder: folderOrder, quickAccess: quickAccess };
}

function countTotalBookmarks(links) {
  if (!links || typeof links !== 'object') return 0;
  var sum = 0;
  for (var k in links) { if (links.hasOwnProperty(k) && Array.isArray(links[k])) sum += links[k].length; }
  return sum;
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  return sheet;
}

function formatHeaderRow(sheet, numCols, bgColor) {
  try {
    var range = sheet.getRange(1, 1, 1, numCols);
    range.setBackground(bgColor); range.setFontColor('#ffffff');
    range.setFontWeight('bold'); range.setFontFamily('Segoe UI'); range.setFontSize(10);
  } catch (_) {}
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

  var lastBackup = raw && raw.exportDate ? new Date(raw.exportDate).toLocaleString() : 'Never';

৪
Deploy as Web App (মোস্ট ইম্পরট্যান্ট)

Apps Script এডিটরের উপরের ডানদিকের Deploy বাটনে ক্লিক করে New deployment সিলেক্ট করুন:

  1. বাম পাশের গিয়ার (⚙) আইকন থেকে Web app বেছে নিন।
  2. Description: লিখুন Bookly Pro Live Sync
  3. Execute as: সিলেক্ট করুন Me (আপনার ইমেইল)
  4. Who has access: Anyone (অবশ্যই Anyone সিলেক্ট করবেন, অন্যথায় এক্সটেনশন সংযোগ করতে পারবে না)
⚠️ অনুমতি দেওয়া (Authorize Access): Deploy বাটনে ক্লিক করার পর Google একটি পারমিশন উইন্ডো দেখাতে পারে। "Authorize access" > আপনার Google Account > "Advanced" > "Go to Bookly Pro Backup (unsafe)" > "Allow" করে দিন।

ডিপ্লয় সফল হলে একটি Web app URL দেওয়া হবে। সেটি Copy করুন।

(URL দেখতে এমন হবে: https://script.google.com/macros/s/.../exec)
